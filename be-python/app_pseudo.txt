DEFINE FUNCTION convert_png_to_jpeg(filepath):

    IF not filepath.lower().endswith(".png"):

        RETURN filepath

    TRY:

        SET image TO Image.open(filepath).convert("RGB")

        SET new_path TO filepath.replace(".png", ".jpg")

        image.save(new_path, "JPEG")

        RETURN new_path

    except Exception as e:

        logger.error(f"Gagal mengonversi gambar PNG: {e}")

        RETURN filepath

@app.route('/api/transcribe', methods=['POST'])

DEFINE FUNCTION transcribe_audio():

    IF 'audio' not IN request.files:

        RETURN jsonify({"error": "No audio file provided"}), 400



    SET audio_file TO request.files['audio']

    IF audio_file.filename EQUALS '':

        RETURN jsonify({"error": "Empty filename"}), 400



    SET device_id TO request.form.get('device_id')

    SET session_id TO request.form.get('session_id')



    IF not device_id or not session_id:

        RETURN jsonify({"error": "Device ID and Session ID are required"}), 400



    TRY:

        SET temp_folder TO get_temp_upload_folder(device_id, 'audio')

        SET filename TO f"audio_{datetime.now().strftime('%Y%m%d_%H%M%S')}_{str(uuid.uuid4())[:8]}.wav"

        SET filepath TO os.path.join(temp_folder, filename)

        audio_file.save(filepath)



        SET validation TO validate_audio_file(filepath)

        IF validation.get('error'):

            RETURN jsonify(validation), 400



        # Deteksi bahasa terlebih dahulu

        SET audio TO whisper.load_audio(filepath)

        SET audio TO whisper.pad_or_trim(audio)

        SET mel TO whisper.log_mel_spectrogram(audio).to(WHISPER_MODEL.device)

        SET _, probs TO WHISPER_MODEL.detect_language(mel)

        SET detected_lang_code TO max(probs, key=probs.get)

        SET detected_lang_prob TO round(probs[detected_lang_code] * 100, 2)



        # Transkripsi dengan asumsi konteks bahasa lokal

        SET result TO WHISPER_MODEL.transcribe(

            filepath,

            language="id",

            task="transcribe",

            initial_prompt="Bahasa Indonesia digunakan di percakapan ini."

        )



        SET transcription TO result.get("text", "").strip()

        IF not transcription:

            RETURN jsonify({"error": "No speech detected"}), 400



        # Tambahkan pesan ke database

        SET message TO Message(

            id=str(uuid.uuid4()),

            session_id=session_id,

            device_id=device_id,

            content=transcription,

            role='user',

            timestamp=int(datetime.now().timestamp() * 1000),

            audio_path=f"/uploads/temp/{device_id}/audio/{filename}"

        )

        db.session.add(message)



        SET session TO db.session.get(Session, session_id)

        IF session:

            SET session.updated_at TO int(datetime.now().timestamp() * 1000)



        db.session.commit()



        # Jawaban dari DeepSeek

        SET response_text TO get_deepseek_response(

            transcription,

            session_id=session_id,

            device_id=device_id

        )

        SET assistant_message TO Message(

            id=str(uuid.uuid4()),

            session_id=session_id,

            device_id=device_id,

            content=response_text,

            role="assistant",

            timestamp=int(datetime.now().timestamp() * 1000)

        )

        db.session.add(assistant_message)

        db.session.commit()



        RETURN jsonify({

            "status": "success",

            "detected_language": detected_lang_code,

            "language_confidence": detected_lang_prob,

            "transcription": transcription,

            "response": response_text,

            "audio_url": f"/uploads/temp/{device_id}/audio/{filename}"

        })



    except Exception as e:

        logging.error(f"Transcription error: {str(e)}")

        RETURN jsonify({"error": "Audio processing failed"}), 500



DEFINE FUNCTION validate_audio_file(filepath):

    """Validasi format dan durasi audio"""

    TRY:

        # Cek apakah file ada

        IF not os.path.exists(filepath):

            RETURN {"error": "File not found"}

            

        # Cek ukuran file minimal (100 bytes)

        IF os.path.getsize(filepath) < 100:

            RETURN {"error": "File too small (corrupted?)"}

        

        # Validasi dengan ffprobe

        SET result TO subprocess.run([

            'ffprobe', '-v', 'error',

            '-show_entries', 'stream=codec_type,sample_rate,channels',

            '-of', 'json',

            filepath

        ], capture_output=True, text=True)

        

        IF result.RETURNcode != 0:

            RETURN {"error": "Invalid audio file", "details": result.stderr}

            

        # Parse output ffprobe

        SET probe_data TO json.loads(result.stdout)

        SET streams TO probe_data.get('streams', [])

        

        IF not any(s.get('codec_type') EQUALS 'audio' FOR s IN streams):

            RETURN {"error": "No audio stream found"}

            

        # Cek sample rate dan channels

        FOR stream IN streams:

            IF stream.get('codec_type') EQUALS 'audio':

                IF int(stream.get('sample_rate', 0)) < 8000:

                    RETURN {"error": "Sample rate too low (min 8kHz)"}

                IF int(stream.get('channels', 0)) < 1:

                    RETURN {"error": "No audio channels"}

                    

        RETURN {"valid": True}

        

        

    except json.JSONDecodeError:

        RETURN {"error": "Invalid JSON response from ffprobe"}

    except Exception as e:

        RETURN {"error": f"Validation error: {str(e)}"}



@app.route("/api/analyze/image", methods=["POST"])

DEFINE FUNCTION analyze_image():

    from datetime IMPORT datetime

    IMPORT uuid

    IMPORT tempfile



    SET device_id TO request.form.get("device_id")

    SET session_id TO request.form.get("session_id")

    SET note TO request.form.get("note", "").strip()



    IF not device_id:

        RETURN jsonify({"error": "Device ID is required"}), 400



    IF "file" not IN request.files:

        RETURN jsonify({"error": "No file uploaded"}), 400



    # Simpan file gambar

    SET file TO request.files["file"]

    SET filename TO secure_filename(file.filename)

    SET folder TO os.path.join(tempfile.gettempdir(), device_id, "images")

    os.makedirs(folder, exist_ok=True)

    SET filepath TO os.path.join(folder, filename)

    file.save(filepath)



    SET filepath TO convert_png_to_jpeg(filepath)

    logger.info(f"✅ File disimpan di: {filepath}")

    logger.info(f"📏 Ukuran file: {os.path.getsize(filepath)} bytes")

    logger.info(f"📂 File ada?: {'Ya' IF os.path.exists(filepath) else 'Tidak'}")



    TRY:

        # Jalankan prediksi

        SET result TO model.predict(filepath).json()

        logger.info(f"📊 Hasil prediksi mentah dari Roboflow: {result}")

        SET predictions TO result.get("predictions", [])



        IF not predictions:

            SET explanation TO "Tidak terdeteksi penyakit pada gambar tersebut. Pastikan gambar jelas dan fokus pada bagian daun atau tanaman yang bermasalah."

            RETURN jsonify({

                "detected": None,

                "confidence": 0.0,

                "explanation": explanation,

                "image_url": f"/uploads/temp/{device_id}/images/{os.path.basename(filepath)}"

            })



        # Filter prediksi yang valid

        SET filtered_preds TO [p FOR p IN predictions IF "confidence" IN p and "class" IN p]

        IF not filtered_preds:

            SET explanation TO "Hasil prediksi tidak lengkap atau tidak valid."

            RETURN jsonify({

                "detected": None,

                "confidence": 0.0,

                "explanation": explanation,

                "image_url": f"/uploads/temp/{device_id}/images/{os.path.basename(filepath)}"

            })



        # Pilih prediksi dengan confidence tertinggi

        SET top_pred TO max(filtered_preds, key=lambda p: p.get("confidence", 0))

        SET label TO top_pred.get("class", "Tidak diketahui")

        SET confidence TO round(top_pred.get("confidence", 0) * 100, 2)



        # Buat prompt untuk LLM

        IF note:

            SET prompt TO f"Apa itu penyakit '{label}' pada tanaman dan bagaimana cara mengatasinya? Berikut kondisi tambahan dari pengguna: {note}"

        ELSE:

            SET prompt TO f"Apa itu penyakit '{label}' pada tanaman dan bagaimana cara mengatasinya?"



        SET explanation TO get_deepseek_response(prompt)



        SET now TO int(datetime.now().timestamp() * 1000)

        SET session TO db.session.get(Session, session_id) IF session_id else None



        IF session:

            # Simpan hanya note + gambar sebagai pesan user

            db.session.add(Message(

                id=str(uuid.uuid4()),

                session_id=session_id,

                device_id=device_id,

                content=note IF note else "",

                role="user",

                timestamp=now,

                image_path=f"/uploads/temp/{device_id}/images/{os.path.basename(filepath)}"

            ))



            # Simpan respons LLM jika belum duplikat

            SET existing TO db.session.query(Message).filter_by(

                session_id=session_id,

                content=explanation,

                role="assistant"

            ).first()



            IF not existing:

                db.session.add(Message(

                    id=str(uuid.uuid4()),

                    session_id=session_id,

                    device_id=device_id,

                    content=explanation,

                    role="assistant",

                    timestamp=now + 1

                ))



            SET session.updated_at TO now + 1

            db.session.commit()



        RETURN jsonify({

            "detected": label,

            "confidence": confidence,

            "explanation": explanation,

            "image_url": f"/uploads/temp/{device_id}/images/{os.path.basename(filepath)}"

        })



    except Exception as e:

        logger.error(f"Image analysis error: {str(e)}")

        RETURN jsonify({"error": str(e)}), 500

DEFINE FUNCTION map_weather_condition(weather_main):

    """Map OpenWeather conditions to our frontend conditions"""

    SET weather_main TO weather_main.lower()

    

    IF any(x IN weather_main FOR x IN ['clear', 'sun']):

        RETURN 'sunny'

    ELSEIF any(x IN weather_main FOR x IN ['cloud', 'fog', 'mist', 'haze']):

        RETURN 'cloudy'

    ELSEIF any(x IN weather_main FOR x IN ['rain', 'drizzle', 'shower', 'thunder', 'storm']):

        RETURN 'rainy'

    ELSE:

        RETURN 'cloudy'  # Default



DEFINE FUNCTION get_openweather_data(lat, lon):

    """Fetch weather data from OpenWeather API"""

    TRY:

        SET url TO f"https://api.openweathermap.org/data/2.5/weather?lat={lat}&lon={lon}&appid={OPENWEATHER_API_KEY}&units=metric&lang=id"

        SET response TO requests.get(url)

        response.raise_for_status()

        RETURN response.json()

    except requests.exceptions.RequestException as e:

        logger.error(f"OpenWeather API error: {e}")

        RETURN None



DEFINE FUNCTION get_farming_advice(weather_main):

    """Get farming advice based on weather condition"""

    SET weather_main TO weather_main.lower()

    

    IF any(x IN weather_main FOR x IN ['clear', 'sun']):

        RETURN "Cocok untuk panen atau pengeringan hasil panen"

    ELSEIF any(x IN weather_main FOR x IN ['cloud', 'fog', 'mist', 'haze']):

        RETURN "Baik untuk menanam bibit atau penyemprotan pestisida"

    ELSEIF any(x IN weather_main FOR x IN ['rain', 'drizzle', 'shower']):

        RETURN "Hindari pemupukan dan penyemprotan pestisida"

    ELSEIF any(x IN weather_main FOR x IN ['thunder', 'storm']):

        RETURN "Pastikan drainase lahan baik untuk mencegah genangan"

    ELSE:

        RETURN "Pantau kondisi tanaman secara berkala"

DEFINE FUNCTION correct_user_question(raw_question):

    TRY:

        SET correction_payload TO {

            "model": "deepseek-chat",

            "messages": [

                {"role": "system", "content": "Anda adalah asisten pintar yang tugasnya memperbaiki pertanyaan pengguna yang memiliki kesalahan ketik atau tidak jelas. Kembalikan versi pertanyaan yang lebih jelas, tanpa penjelasan tambahan."},

                {"role": "user", "content": raw_question}

            ],

            "temperature": 0.3,

            "max_tokens": 200

        }



        SET headers TO {

            "Authorization": f"Bearer {DEEPSEEK_API_KEY}",

            "Content-Type": "application/json"

        }



        SET response TO requests.post(

            "https://api.deepseek.com/v1/chat/completions",

            headers=headers,

            json=correction_payload

        )



        IF response.status_code EQUALS 200:

            SET result TO response.json()

            SET corrected TO result['choices'][0]['message']['content']

            RETURN corrected.strip()

        ELSE:

            logger.warning(f"DeepSeek correction failed, using original: {response.text}")

            RETURN raw_question

    except Exception as e:

        logger.error(f"Correction error: {e}")

        RETURN raw_question



DEFINE FUNCTION get_deepseek_response(prompt, session_id=None, device_id=None):

    TRY:

        SET session TO db.session.get(Session, session_id) IF session_id else None

        SET topic TO session.topic.lower() IF session and session.topic else None

        SET topic_instruction TO f"\n\nTopik yang sedang dibahas adalah **{topic}**.\n" IF topic else ""



        # Koreksi pertanyaan

        SET corrected_prompt TO correct_user_question(prompt)



        # Ambil 2 pesan user terakhir yang sesuai topik

        SET history TO []

        IF session_id and device_id:

            SET messages TO Message.query.filter_by(session_id=session_id, device_id=device_id)\

                                    .order_by(Message.timestamp.desc())\

                                    .all()

            FOR msg IN messages:

                IF msg.role EQUALS 'user' and (not topic or topic IN msg.content.lower()):

                    history.append({"role": "user", "content": msg.content})

                    IF len(history) >= 2:

                        break

        SET history TO list(reversed(history))  # Urutkan lama ke baru

        history.append({"role": "user", "content": corrected_prompt})



        # Prompt lama + sisipan topik

        SET system_prompt TO f"""Anda adalah Asisten Pertanian PeTaniku yang ahli di bidang:

- Pertanian dan perkebunan

- Cuaca dan iklim untuk pertanian

- Pengelolaan tanaman dan tanah

- Teknologi pertanian

{topic_instruction}

Bantu pengguna dengan:

1. Berikan jawaban singkat dan jelas untuk pertanyaan pertanian

2. Jika pertanyaan di luar topik, jawab dengan sopan:

   "Maaf, saya hanya dapat membantu tentang pertanian. Ada yang bisa saya bantu terkait tanaman, cuaca pertanian, atau hal terkait?"



Gaya respons:

- Gunakan bahasa sederhana dan praktis

- Format jelas dengan paragraf terpisah

- Hindari jargon teknis berlebihan"""



        SET payload TO {

            "model": "deepseek-chat",

            "messages": [{"role": "system", "content": system_prompt}] + history,

            "temperature": 0.5,

            "max_tokens": 1000

        }



        SET headers TO {

            "Authorization": f"Bearer {DEEPSEEK_API_KEY}",

            "Content-Type": "application/json"

        }



        SET response TO requests.post("https://api.deepseek.com/v1/chat/completions", headers=headers, json=payload)



        IF response.status_code EQUALS 200:

            SET result TO response.json()

            RETURN result['choices'][0]['message']['content']

        ELSE:

            logger.error(f"DeepSeek API error: {response.text}")

            RETURN "Maaf, saya tidak bisa memberikan jawaban saat ini."



    except Exception as e:

        logger.error(f"Error getting DeepSeek response: {e}")

        RETURN "Maaf, terjadi kesalahan dalam memproses permintaan Anda."



DEFINE FUNCTION extract_topic_from_question(question):

    TRY:

        SET response TO get_deepseek_response(

            f"Apa topik utama dari pertanyaan ini? '{question}'. Beri jawaban singkat, maksimal 3 kata."

        )

        RETURN response.strip()

    except Exception as e:

        logger.error(f"Gagal ekstrak topik: {e}")

        RETURN None
